  <h1>🏠 屋企附近巴士實時班次</h1>

  <div id="bus-container">載入中...</div>


  <script>

    // ⚠️ 請替換成你屋企附近車站的 Stop ID

    const TARGET_STOP_ID = "YOUR_STOP_ID_HERE"; 


    async function fetchETA() {

      try {

        const response = await fetch(`https://data.etabus.gov.hk/v1/transport/kmb/stop-eta/${TARGET_STOP_ID}`);

        const result = await response.json();

        

        const container = document.getElementById('bus-container');

        container.innerHTML = '';


        // 過濾並只取前幾班車

        const list = result.data || [];

        if (list.length === 0) {

          container.innerHTML = '<p style="text-align:center;">暫無班次資料</p>';

          return;

        }


        // 整理並排序班次

        list.slice(0, 6).forEach(item => {

          if (!item.eta) return;


          // 計算剩餘時間（分鐘）

          const etaTime = new Date(item.eta);

          const now = new Date();

          const diffMinutes = Math.round((etaTime - now) / 60000);


          let etaDisplay = '';

          if (diffMinutes <= 0) {

            etaDisplay = '即將到達';

          } else {

            etaDisplay = `${diffMinutes} 分鐘`;

          }


          // 生成卡片 HTML

          const card = document.createElement('div');

          card.className = 'card';

          card.innerHTML = `

            <div class="route-info">

              <span class="route-number">${item.route}</span>

              <span class="destination">往 ${item.dest_tc}</span>

            </div>

            <div class="eta">${etaDisplay}</div>

          `;

          container.appendChild(card);

        });


      } catch (error) {

        console.error('獲取數據失敗:', error);

      }

    }


    // 首次載入

    fetchETA();


    // 每 30 秒自動刷新一次

    setInterval(fetchETA, 30000);

  </script>
